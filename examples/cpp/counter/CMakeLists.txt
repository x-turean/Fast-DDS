cmake_minimum_required(VERSION 3.22)

project(fastdds_counter_example VERSION 1 LANGUAGES CXX)

if(NOT fastcdr_FOUND)
    find_package(fastcdr 2 REQUIRED)
endif()

if(NOT fastdds_FOUND)
    find_package(fastdds 3 REQUIRED)
endif()

message(STATUS "Configuring counter example...")

add_executable(counter_publisher CounterPublisher.cpp)
target_link_libraries(counter_publisher fastdds fastcdr)

add_executable(counter_subscriber CounterSubscriber.cpp)
target_link_libraries(counter_subscriber fastdds fastcdr)
